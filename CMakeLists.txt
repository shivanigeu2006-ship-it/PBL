cmake_minimum_required(VERSION 3.10)
project(SmartStock CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
include_directories(include src)
find_package(Threads REQUIRED)
add_executable(smartstock src/main.cpp)
target_link_libraries(smartstock Threads::Threads)
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core Threads::Threads)
if(WIN32)
  target_link_libraries(smartstock ws2_32 mswsock)
  target_link_libraries(test_core ws2_32 mswsock)
  target_compile_definitions(smartstock PRIVATE _WIN32_WINNT=0x0A00)
  if(MSVC)
    target_compile_options(smartstock PRIVATE /utf-8)
  endif()
endif()
